% ------------------------------------------------------------------------------
% Figure 3: HEDO Coordinate-Momentum Mechanism (Independent Single Column)
% ------------------------------------------------------------------------------
\newcommand{\HEDOMechanismFigure}{
\begin{tikzpicture}[node distance=5mm, every node/.style={font=\footnotesize}]
\node[hvinput, minimum width=72mm] (q)
{\textbf{Coordinate State}\\$\mathbf{q}_0 \in \mathbb{R}^{B \times L \times d_{\text{model}}}$};

\node[hvstate, below=of q, minimum width=72mm] (p)
{\textbf{Momentum Initialization}\\$\mathbf{p}_0 = \tanh(\mathbf{W}_p \mathbf{q}_0)$};

\node[hvmethod, below=of p, minimum width=76mm] (m)
{\textbf{Dissipative Momentum Update}\\$\mathbf{p}_{t+1} = \mathbf{p}_t(1 - \beta\Delta t) - \Delta t \tanh(\mathbf{W}_q \mathbf{q}_t)$\\[2pt]
\scriptsize Damping Factor: $(1 - \beta\Delta t) = 0.995 < 1\quad(\beta=0.05, \Delta t=0.1)$};

\node[hvmethod, below=of m, minimum width=76mm] (c)
{\textbf{Discrete Coordinate Update}\\$\mathbf{q}_{t+1} = \mathbf{q}_t + \gamma\Delta t\,\mathbf{p}_{t+1}$\\[2pt]
\scriptsize Step Scale: $\gamma = 0.1,\;\Delta t = 0.1$};

\node[hvenc, below=of c, minimum width=72mm] (ln)
{\textbf{Layer Normalization}\\$\text{HEDO}(\mathbf{q}) = \text{LayerNorm}(\mathbf{q}_{t+1})$};

\foreach \a/\b in {q/p,p/m,m/c,c/ln} \draw[hvarrow] (\a) -- (\b);
\end{tikzpicture}
}
